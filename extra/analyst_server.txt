"""
analyst_server.py - Lab 3: Serving an agent over A2A

This publishes the Company Analyst as a real A2A agent:
  - an Agent Card at /.well-known/agent-card.json (discovery)
  - a JSON-RPC 2.0 endpoint at / (communication)
"""

import uvicorn
from starlette.applications import Starlette

from a2a.server.request_handlers import DefaultRequestHandler
from a2a.server.routes import create_agent_card_routes, create_jsonrpc_routes
from a2a.server.tasks import InMemoryTaskStore
from a2a.types import AgentCapabilities, AgentCard, AgentInterface, AgentSkill

from analyst_executor import AnalystExecutor

# ---------------------------------------------------------------
# The Agent Card - how other agents discover what this agent can do
# ---------------------------------------------------------------

skill = AgentSkill(
    id="company_analysis",
    name="Company Analysis",
    description=("Answers questions about Acme Corp, Globex, and Initech: "
                 "profiles, financials, recent news, and exact calculations."),
    tags=["companies", "finance", "news", "analysis"],
    examples=["What is Globex's revenue per employee?",
              "What's the latest news about Initech?"],
    input_modes=["text/plain"],
    output_modes=["text/plain"],
)

card = AgentCard(
    name="Company Analyst Agent",
    description="A tool-using analyst agent for company research and financial questions.",
    version="1.0.0",
    default_input_modes=["text/plain"],
    default_output_modes=["text/plain"],
    capabilities=AgentCapabilities(streaming=True),
    supported_interfaces=[
        AgentInterface(protocol_binding="JSONRPC", url="http://127.0.0.1:9999",
                       protocol_version="1.0"),
    ],
    skills=[skill],
)

# ---------------------------------------------------------------
# The server - SDK request handler + routes on a Starlette app
# ---------------------------------------------------------------

handler = DefaultRequestHandler(
    agent_executor=AnalystExecutor(),
    task_store=InMemoryTaskStore(),
    agent_card=card,
)

routes = []
routes.extend(create_agent_card_routes(card))       # serves the Agent Card
routes.extend(create_jsonrpc_routes(handler, "/"))  # serves JSON-RPC methods

app = Starlette(routes=routes)


if __name__ == "__main__":
    print("Starting Company Analyst Agent on http://127.0.0.1:9999")
    print("Agent Card: http://127.0.0.1:9999/.well-known/agent-card.json")
    uvicorn.run(app, host="127.0.0.1", port=9999)
